# Список встроенных правил валидации Laravel

## 📌 Обязательность и присутствие полей
```php
$rules = [
'field_1'  => 'required',               // Поле обязательно к заполнению, не может быть пустым или null
'field_2'  => 'nullable',               // Поле может быть null (сбрасывает строгие проверки для пустых значений)
'field_3'  => 'sometimes',              // Проверка запускается только в том случае, если поле присутствует в массиве данных
'field_4'  => 'present',                // Поле обязательно должно присутствовать в данных, даже если оно пустое или null
'field_5'  => 'missing',                // Поле обязательно должно отсутствовать в запросе
'field_6'  => 'missing_if:status,1',     // Поле должно отсутствовать, если другое поле status равно 1
'field_7'  => 'missing_unless:status,1', // Поле должно отсутствовать, кроме случая, когда status равен 1
'field_8'  => 'missing_with:field_a',    // Поле должно отсутствовать, если присутствует поле field_a
'field_9'  => 'missing_with_all:a,b',    // Поле должно отсутствовать, если присутствуют оба поля a и b
'field_10' => 'required_if:status,1',    // Обязательно, если поле status равно 1
'field_11' => 'required_unless:status,1',// Обязательно, если только поле status НЕ равно 1
'field_12' => 'required_with:field_a',   // Обязательно, если присутствует (не null) поле field_a
'field_13' => 'required_with_all:a,b',   // Обязательно, если присутствуют все перечисленные поля (a и b)
'field_14' => 'required_without:field_a',// Обязательно, если поле field_a отсутствует или пустое
'field_15' => 'required_without_all:a,b',// Обязательно, если отсутствуют все перечисленные поля (a и b)
'field_16' => 'accepted',               // Должно быть "yes", "on", 1, "1", true или "true" (для чекбоксов оферты)
'field_17' => 'accepted_if:status,1',   // Должно быть принято (принято=true), если поле status равно 1
'field_18' => 'declined',               // Должно быть "no", "off", 0, "0", false или "false"
'field_19' => 'declined_if:status,1',   // Должно быть отклонено, если поле status равно 1
'field_20' => 'filled',                 // Если поле присутствует, оно не должно быть пустым
];
```


## 📌 Типы данных и форматы
```php
$rules = [
'field_21' => 'string',                 // Должно быть строкой
'field_22' => 'integer',                // Должно быть целым числом
'field_23' => 'numeric',                // Должно быть числом (целым или с плавающей точкой)
'field_24' => 'boolean',                // Должно логическим типом (или значениями true, false, 1, 0, "1", "0")
'field_25' => 'array',                  // Должно быть массивом PHP
'field_26' => 'array:name,email',       // Должно быть массивом, содержащим строго указанные ключи (name, email)
'field_27' => 'json',                   // Должно быть валидной JSON-строкой
'field_28' => 'alpha',                  // Должно состоять строго из букв
'field_29' => 'alpha_dash',             // Должно состоять из букв, цифр, дефисов (-) и знаков подчеркивания (_)
'field_30' => 'alpha_num',              // Должно состоять только из букв и цифр
'field_31' => 'ascii',                  // Должно состоять только из 7-битных ASCII символов
'field_32' => 'ulid',                   // Должно быть валидным идентификатором ULID
'field_33' => 'uuid',                   // Должно быть валидным идентификатором UUID
];
```


## 📌 Числа, длина и размеры
```php
$rules = [
'field_34' => 'size:5',                 // Строго фиксированный размер (5 символов для строк, число 5 для цифр, 5 элементов для массивов, 5 КБ для файлов)
'field_35' => 'min:3',                  // Минимальный размер (от 3 символов, число >= 3, от 3 элементов, от 3 КБ)
'field_36' => 'max:255',                // Максимальный размер (до 255 символов, число <= 255, до 255 элементов, до 255 КБ)
'field_37' => 'between:10,20',          // Размер должен быть строго в диапазоне от 10 до 20 включительно
'field_38' => 'digits:4',               // Должно быть числом и иметь ровно 4 цифры в длину
'field_39' => 'digits_between:2,5',     // Должно быть числом и иметь длину от 2 до 5 цифр
'field_40' => 'decimal:2',              // Должно быть числом с плавающей точкой, имеющим ровно 2 знака после запятой
'field_41' => 'decimal:1,3',            // Число с плавающей точкой, имеющее от 1 до 3 знаков после запятой
];
```


## 📌 Сравнения и логика
```php
$rules = [
'field_42' => 'same:password',          // Значение поля должно полностью совпадать со значением поля password
'field_43' => 'different:old_password', // Значение поля должно отличаться от значения поля old_password
'field_44' => 'confirmed',              // Требует поле {field}_confirmation. Например, для password нужно поле password_confirmation
'field_45' => 'gt:age',                 // Должно быть строго больше (>), чем значение поля age
'field_46' => 'gte:age',                // Должно быть больше или равно (>=), чем значение поля age
'field_47' => 'lt:age',                 // Должно быть строго меньше (<), чем значение поля age
'field_48' => 'lte:age',                // Должно быть меньше или равно (<=), чем значение поля age
];
```


## 📌 Работа со списками и соответствиями
```php
$rules = [
'field_49' => 'in:admin,editor,user',   // Значение должно строго совпадать с одним из указанных вариантов
'field_50' => 'not_in:root,moderator',  // Значение НЕ должно совпадать ни с одним из указанных вариантов
'field_51' => 'in_array:another_list.*',// Значение должно существовать в массиве другого поля `another_list`
'field_52' => 'distinct',               // При валидации массива (`array.*`) требует, чтобы все его элементы были уникальными
'field_53' => 'distinct:strict',        // Уникальность элементов в массиве с использованием строгого сравнения (===)
'field_54' => 'distinct:ignore_case',   // Уникальность элементов в массиве без учета регистра букв
];
```


## 📌 Сетевые протоколы и интернет
```php

'field_58' => 'active_url',             // Должен быть валидным URL-адресом с активной DNS-записью (A или AAAA)
'field_59' => 'ip',                     // Должно быть валидным IP-адресом (IPv4 или IPv6)
'field_60' => 'ipv4',                   // Должно быть строго валидным IPv4-адресом
'field_61' => 'ipv6',                   // Должно быть строго валидным IPv6-адресом
'field_62' => 'mac_address',            // Должно быть валидным MAC-адресом
];
```


## 📌 Дата и время
```php
$rules = [
'field_63' => 'date',                   // Должно быть корректной датой согласно PHP-функции strtotime()
'field_64' => 'date_format:Y-m-d',      // Должно строго соответствовать указанному формату даты (например, 2026-07-16)
'field_65' => 'before:tomorrow',        // Дата должна быть раньше указанной (можно передавать строки для strtotime или имя другого поля)
'field_66' => 'before_or_equal:today',  // Дата должна быть раньше или равна указанной дате
'field_67' => 'after:2026-01-01',       // Дата должна быть позже указанной даты
'field_68' => 'after_or_equal:start_dt',// Дата должна быть позже или равна дате из поля start_dt
'field_69' => 'timezone',               // Должно быть валидным именем часового пояса (например, UTC, Europe/Moscow)
'field_70' => 'timezone:Africa',        // Должно быть валидным часовым поясом, принадлежащим к конкретному региону (Africa)
];
```


## 📌 База данных (Database)
```php
$rules = [
'field_71' => 'exists:users',                 // Значение должно существовать в таблице users в колонке с именем текущего поля
'field_72' => 'exists:users,email',           // Значение должно существовать в таблице users в конкретной колонке email
'field_73' => 'unique:users',                 // Значение должно быть уникальным в таблице users (не должно там существовать)
'field_74' => 'unique:users,email_address',   // Должно быть уникальным в таблице users в конкретной колонке email_address
];
```


## 📌 Файлы и изображения
```php
$rules = [
'field_75' => 'file',                                // Должно быть успешно загруженным файлом
'field_76' => 'image',                               // Файл должен быть изображением (jpg, jpeg, png, bmp, gif, svg, или webp)
'field_77' => 'mimes:jpg,png,pdf',                   // Файл должен иметь один из указанных MIME-типов/расширений
'field_78' => 'mimetypes:video/avi,video/mpeg',      // Файл должен строго соответствовать указанным MIME-типам контента
'field_79' => 'extensions:jpg,png',                  // Проверка файла строго по его расширению, без проверки содержимого (MIME)
'field_80' => 'dimensions:min_width=100,min_height=200', // Картинка должна иметь параметры: минимум 100px в ширину и 200px в высоту
];
```


## 📌 Безопасность и регулярные выражения
```php
$rules = [
'field_81' => 'regex:/^[a-z]+$/i',      // (Осторожно!) Проверка на соответствие регулярному выражению. *Символ | внутри регулярки сломает строковый синтаксис!*
'field_82' => 'not_regex:/[0-9]/',      // Значение НЕ должно соответствовать указанному регулярному выражению
'field_83' => 'current_password',       // Значение должно совпадать с текущим паролем авторизованного пользователя
'field_84' => 'current_password:api',   // Проверка текущего пароля с использованием конкретного guard (например, api)
];
```
